module SA_Cache(
input [31:0] PC,
input CLK,
input update,
input SA_Cache_WE,
input SA_Cache_rden,
input logic [31:0] rw0,
input logic [31:0] rw1,
input logic [31:0] rw2,
input logic [31:0] rw3,
input logic [31:0] DIN2,
output logic [31:0] ww0,
output logic [31:0] ww1,
output logic [31:0] ww2,
output logic [31:0] ww3,
output logic [31:0] rd,
output logic hit,
output logic miss,
output logic wb_mem,
output logic [31:0] wb_adr
);
//parameter NUM_BLOCKS = 16;
parameter BLOCKS_PER_SET = 4;
parameter NUM_SETS = 4;
parameter BLOCK_SIZE = 4;
parameter SET_INDEX_SIZE = 2;
parameter WORD_OFFSET_SIZE = 2;
parameter BYTE_OFFSET = 2;
parameter TAG_SIZE = 32 - SET_INDEX_SIZE - WORD_OFFSET_SIZE - BYTE_OFFSET;

typedef struct {
    logic [31:0] data[4];
    logic [TAG_SIZE-1:0] tag;
    logic valid;
    logic dirty;
    logic [1:0] lru;
} cache_block;

cache_block cache[NUM_SETS][BLOCKS_PER_SET];

initial begin
    // iterate through sets
    for(int s = 0; s < NUM_SETS; s++) begin
        // iterate through blocks
        for(int b = 0; b < BLOCKS_PER_SET; b++) begin //initializing RAM to 0
            cache[s][b].tag = 0;
            cache[s][b].valid = 0;
            cache[s][b].dirty = 0;
            cache[s][b].lru = 0;
            // iterate through each word withing block
            for(int w = 0; w < BLOCK_SIZE; w++)
                cache[s][b].data[w] = 0;
        end
    end
end

logic [1:0] block_index, lru_index;
logic found_invalid;
logic [TAG_SIZE-1:0] pc_tag_comb; // Renamed to avoid conflict with module input if any
logic [SET_INDEX_SIZE-1:0] set_index_comb;
logic [WORD_OFFSET_SIZE-1:0] word_offset_comb;


assign word_offset_comb = PC[WORD_OFFSET_SIZE+BYTE_OFFSET-1:BYTE_OFFSET]; 
assign set_index_comb = PC[SET_INDEX_SIZE+WORD_OFFSET_SIZE+BYTE_OFFSET-1:WORD_OFFSET_SIZE+BYTE_OFFSET];
assign pc_tag_comb = PC[31:(SET_INDEX_SIZE+WORD_OFFSET_SIZE+BYTE_OFFSET)];
assign miss = !hit;

// determine if hit
always_comb begin
    hit = 1'b0;
    block_index = 2'bxx; // Default to unknown
    for(int way = 0; way < BLOCKS_PER_SET; way++) begin
        if (cache[set_index_comb][way].valid && cache[set_index_comb][way].tag == pc_tag_comb) begin
            hit = 1'b1;
            block_index = way;
            break;
        end
    end
end

// logic to determine lru given set index
always_comb begin
    found_invalid = 0;
    lru_index = 2'bxx; // Default to unknown
    // check if there is invalid block
    for(int way = 0; way < BLOCKS_PER_SET; way++) begin
        if (!cache[set_index_comb][way].valid) begin
            lru_index = way;
            found_invalid = 1;
            break;
        end
    end
    
    // if no invalid block then find index for highest lru
    if (!found_invalid) begin
        logic [1:0] max_lru = cache[set_index_comb][0].lru; // Assume way 0 if all are valid and have same LRU
        lru_index = 0; 
        for(int way = 0; way < BLOCKS_PER_SET; way++) begin
            if(cache[set_index_comb][way].lru > max_lru) begin
                max_lru = cache[set_index_comb][way].lru;
                lru_index = way;
            end
        end
    end
end

// check if wb is needed and set wb_mem to 1 if needed and 0 if not
always_comb begin
    wb_mem = 0;
    // check if writing, not a hit, valid bit of lru_way is 1, and dirty bit of lru_way is 1
    // This signal indicates the victim (lru_index) needs to be written back BEFORE it's replaced.
    // This should also apply for read misses that evict a dirty block.
    // For now, keeping original condition, but FSM needs broader WB logic.
    if ((SA_Cache_WE || SA_Cache_rden) && !hit && lru_index != 2'bxx && cache[set_index_comb][lru_index].valid && cache[set_index_comb][lru_index].dirty) begin
        wb_mem = 1;
    end
end

// synchronous read based on read enable
always_ff @ (posedge CLK) begin
    if(SA_Cache_rden && hit) begin // Only provide data if it's a hit
        rd <= cache[set_index_comb][block_index].data[word_offset_comb];
    end
    // On a miss, 'rd' is not driven by this block; FSM handles stall and data validity.
end

// update LRU regardless of writing or reading
always_ff @(posedge CLK) begin
    if(hit && (SA_Cache_WE || SA_Cache_rden)) begin
        if (block_index != 2'bxx) begin // Ensure block_index is valid
            for(int way = 0; way < BLOCKS_PER_SET; way++) begin
                if (way == block_index) begin
                    cache[set_index_comb][way].lru <= 2'b00;
                end else if (cache[set_index_comb][way].valid && cache[set_index_comb][way].lru < (BLOCKS_PER_SET - 1)) begin // Increment others if valid and not max
                    cache[set_index_comb][way].lru <= cache[set_index_comb][way].lru + 1;
                end
            end
        end
    end
end

// handle cache writing and write-back data preparation
always_ff @ (posedge CLK) begin
    // check if writing to cache
    if(SA_Cache_WE) begin
        // check if place being written to is in cache
        if(hit && block_index != 2'bxx) begin // Write Hit
            cache[set_index_comb][block_index].dirty <= 1;
            cache[set_index_comb][block_index].data[word_offset_comb] <= DIN2;
            // Tag and Valid of block_index remain correct, no update needed here for them.
            // LRU is updated in a separate block.
        end
        // if writing but miss (Write Allocate strategy started here, FSM must complete it)
        else if (!hit && lru_index != 2'bxx) begin 
            // If the victim block (lru_index) is valid and dirty, its data and address
            // should be latched for write-back based on 'wb_mem' signal.
            // The FSM will manage the actual write-back using these latched values.
            if (cache[set_index_comb][lru_index].valid && cache[set_index_comb][lru_index].dirty) begin
                // Latch data from the victim block (lru_index) for potential write-back
                ww0 <= cache[set_index_comb][lru_index].data[0];
                ww1 <= cache[set_index_comb][lru_index].data[1];
                ww2 <= cache[set_index_comb][lru_index].data[2];
                ww3 <= cache[set_index_comb][lru_index].data[3];
                wb_adr <= {cache[set_index_comb][lru_index].tag, set_index_comb, {BYTE_OFFSET{1'b0}} }; // Use victim's OLD tag
            end

            // The following lines prepare the lru_index way for the new data (from DIN2 and subsequent 'update')
            // This is part of write-allocate: the word is written, then block is fetched.
            // This specific write of DIN2 might be overwritten by 'update' if FSM doesn't handle it.
            // This remains an architectural point for FSM design.
            cache[set_index_comb][lru_index].dirty <= 1; // Will be dirty due to this write
            cache[set_index_comb][lru_index].valid <= 1; // Will become valid
            cache[set_index_comb][lru_index].data[word_offset_comb] <= DIN2; // CPU data written
            cache[set_index_comb][lru_index].tag <= pc_tag_comb; // New tag for the incoming block
            // LRU for lru_index will be set to 0 by FSM after this miss is serviced, or by a similar rule as hit.
        end
    end
end

// This block handles filling the cache line from memory (rw0-rw3) when 'update' is asserted by FSM
always_ff @ (posedge CLK) begin
    if(update && lru_index != 2'bxx) begin // 'update' comes from FSM after rw0-3 are valid from memory
        cache[set_index_comb][lru_index].data[0] <= rw0;
        cache[set_index_comb][lru_index].data[1] <= rw1;
        cache[set_index_comb][lru_index].data[2] <= rw2;
        cache[set_index_comb][lru_index].data[3] <= rw3;
        cache[set_index_comb][lru_index].valid <= 1;
        cache[set_index_comb][lru_index].tag <= pc_tag_comb; // Tag of the block just fetched
        // If 'update' follows a write miss where DIN2 was already written to data[word_offset_comb],
        // and if rwX contains the whole block from memory, DIN2 is overwritten.
        // This needs FSM coordination. For now, assuming FSM might set rw[word_offset_comb] = DIN2 before update.
        // A clean fill from memory should make the block not dirty initially.
        // If it was a write miss that caused this fill, it will be marked dirty by the SA_Cache_WE logic.
        // So, if this 'update' is for a read miss, dirty should be 0.
        // If this 'update' is for a write miss (where DIN2 was also written), dirty is already 1.
        // For simplicity, let's assume a fill makes it clean, write operation will set dirty.
        if (!SA_Cache_WE || hit) begin // If it's not an ongoing write miss that just wrote DIN2
            cache[set_index_comb][lru_index].dirty <= 0;
        end
        // The LRU for lru_index should be set to 0 after this update (similar to a hit)
        // This is often handled by the same LRU update logic if FSM makes 'update' look like a 'hit' to lru_index
    end
end

endmodule
